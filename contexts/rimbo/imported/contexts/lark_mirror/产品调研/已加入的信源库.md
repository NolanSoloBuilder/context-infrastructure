class ChannelSourceDao(BaseModel):
    source_id: str  # 唯一ID
    name: str  # 展示名
    logo_url: Optional[str] = None  # 信源图标
    canonical_url: Optional[str] = None  # 信源页面
    rss_url: Optional[str] = None  # RSS/Atom 订阅地址
    source_type: str  # website / newsletter / podcast 
    formats: Optional[List[str]] = None  # html / text 
    reading_template: Optional[str] = None  # 默认阅读模版类型
    theme_tags: Optional[List[str]] = None  # 主题标签多个
    audience_fit: Optional[str] = None  # product_manager / investor / founder / researcher
    source_summary: Optional[str] = None  # 一句话描述
    publisher_name: Optional[str] = None  # 信源发布者主体名
    publisher_type: Optional[str] = None  # person / company / platform
    recent_topic_snapshot: Optional[Dict[str, float]] = None  # 最近3条内容主题快照
    snapshot_generated_at: Optional[datetime | str] = None  # 快照生成时间
    authority_signals: Optional[Dict[str, Any]] = None  # 权威性字段
    perspective_type: Optional[str] = None  #  观察角度的类型 
    source_quality_score: Optional[float] = None  # 信源质量分
    last_success_at: Optional[datetime | str] = None  # 最近一次成功抓取时间
    status: str = "active"  # active / inactive   
    seed_source: Optional[str] = None  # team_upload / user_upload
    created_at: Optional[datetime | str] = None
    updated_at: Optional[datetime | str] = None
```

class ChannelFeedDao(BaseModel):
    source_id: str  # 关联信源ID
    item_id: str  # 源内唯一指纹（唯一索引）
    status: str = "NEW"  # 状态
    created_at: Optional[datetime | str] = None  # 创建时间
    updated_at: Optional[datetime | str] = None  # 更新时间

    # --- 前端展示相关 ---
    title: Optional[str] = None  # 条目标题
    link: Optional[str] = None  # 原始链接
    content_type: str  # audio | text | image | video | mixed
    published_at: Optional[datetime | str] = None  # 发布时间（UTC）
    cover_url: Optional[str] = None  # 主封面/首图
    media_urls: Optional[List[str]] = None  # 媒资列表（音频/图片/视频）
    text: Optional[str] = None  # 规范化正文/转写
    summary: Optional[str] = None  # 摘要
    lang: Optional[str] = None  # 语言标识

    # --- 检索/推荐相关 ---
    match_keys: Optional[List[str]] = None  # 匹配用关键字段
    keywords: Optional[List[str]] = None  # 提取/落库关键词
    analysis: Optional[Dict[str, Any]] = None  # LLM分析结果（得分/分类等）
    signals: Optional[List[Dict[str, Any]]] = None  # 结构化信号列表
    fingerprint: Optional[str] = None  # 跨源/全局去重指纹（可选）

    # --- 调试/维护相关 ---
    error: Optional[str | Dict[str, Any]] = None  # 解析/处理错误
    retry_count: Optional[int] = None  # 重试次数
    expires_at: Optional[datetime | str] = None  # 过期时间（TTL/清理）
    raw_payload: Optional[Dict[str, Any]] = None  # 原始解析结果

```
